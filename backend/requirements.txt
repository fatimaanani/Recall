# Uploads module. python-multipart is a hard FastAPI/Starlette requirement
# for parsing multipart/form-data (file uploads + form fields) , without
# it, any route using UploadFile/Form raises a runtime error the moment
# it's called. No file-type-sniffing library (e.g. python-magic) is added
# here: it needs the system libmagic library, which is extra setup friction
# for a graduation project; app/utils/file_validation.py instead checks a
# small set of known magic-byte signatures directly (see that file).
python-multipart==0.0.9

# Media processing module - Whisper transcription stage
# (app/processing/whisper_processor.py). faster-whisper is a CTranslate2-
# based reimplementation of OpenAI's Whisper: same models/accuracy, much
# faster on CPU, and it's what's actually imported (no plain `whisper` /
# `openai-whisper` package is used). Still requires the system FFmpeg
# binary already documented in .env.example , this stage extracts each
# video's audio track with it before handing the result to Whisper.
faster-whisper==1.0.3
requests==2.32.3

# Embedding generation module (app/processing/embedding_processor.py),
# populates transcript_segments.embedding for semantic search. all-MiniLM-
# L6-v2 (the default model, see app/config.py) outputs 384-dim vectors,
# matching the Vector(384) column already defined in
# app/models/transcript_segment.py , any other model swapped in later
# must match that dimension or the column needs its own migration.
# sentence-transformers pulls in torch/transformers/numpy as transitive
# dependencies; there's no lighter-weight option that still runs a real
# sentence-embedding model locally (no external embedding API is used, to
# keep this offline and free, consistent with faster-whisper above).
sentence-transformers==3.0.1
